cmake_minimum_required(VERSION 3.22)
project(madalasa_backend LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Drogon CONFIG REQUIRED)
# Ubuntu's Drogon 1.8 exports the pre-0.8 yaml-cpp target spelling.
# Bind that spelling to the actual imported target, including its SDK path.
if(NOT TARGET yaml-cpp AND TARGET yaml-cpp::yaml-cpp)
  add_library(yaml-cpp ALIAS yaml-cpp::yaml-cpp)
endif()
add_library(madalasa_database src/Database.cc)
target_include_directories(madalasa_database PUBLIC include)
target_link_libraries(madalasa_database PUBLIC Drogon::Drogon)
target_compile_options(madalasa_database PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(madalasa_backend src/main.cc)
target_link_libraries(madalasa_backend PRIVATE madalasa_database)
target_compile_options(madalasa_backend PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_executable(database_config_test tests/database_config_test.cc)
target_link_libraries(database_config_test PRIVATE madalasa_database)
add_test(NAME database_config COMMAND database_config_test)
